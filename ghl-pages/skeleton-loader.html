<script>
  (function () {
    var LOAD_MS = 2200; // pretend the content takes 2.2 seconds to arrive
    var CLIP_S = 5;     // length of the radio clip in seconds

    /* ---------- the 5-second lofi clip, made live in the browser (no audio file) ----------
       Jazzy chords (Dm9 - G13 - Cmaj9 - Am7) on a soft electric-piano tone with a gentle
       tape wobble, a round bass, a lazy swung drum beat at 72 BPM, vinyl crackle and tape hiss. */
    function buildClip(ctx, out, t0) {
      var END = t0 + CLIP_S;
      var BEAT = 60 / 72;
      var hz = function (m) { return 440 * Math.pow(2, (m - 69) / 12); };

      // master volume: quick fade in, gentle fade out, through a warm low-pass and a limiter
      var master = ctx.createGain();
      master.gain.setValueAtTime(0, t0);
      master.gain.linearRampToValueAtTime(0.85, t0 + 0.12);
      master.gain.setValueAtTime(0.85, END - 0.9);
      master.gain.linearRampToValueAtTime(0, END);
      var warm = ctx.createBiquadFilter();
      warm.type = "lowpass"; warm.frequency.value = 2600; warm.Q.value = 0.4;
      var limit = ctx.createDynamicsCompressor();
      limit.threshold.value = -10; limit.ratio.value = 6; limit.attack.value = 0.003; limit.release.value = 0.2;
      master.connect(warm); warm.connect(limit); limit.connect(out);

      // white noise, reused by drums and hiss
      var noise = ctx.createBuffer(1, Math.floor(ctx.sampleRate * (CLIP_S + 0.5)), ctx.sampleRate);
      var nd = noise.getChannelData(0);
      for (var i = 0; i < nd.length; i++) nd[i] = Math.random() * 2 - 1;

      // tape wobble: a slow wave that bends the chord pitch a few cents
      var wow = ctx.createOscillator(); wow.frequency.value = 0.55;
      var wowDepth = ctx.createGain(); wowDepth.gain.value = 9;
      wow.connect(wowDepth); wow.start(t0); wow.stop(END + 0.1);

      // ---- chords ----
      var chords = [
        { at: 0,    bass: 38, notes: [53, 57, 60, 64] },     // Dm9   (bass D)
        { at: 1.25, bass: 43, notes: [53, 59, 64, 69] },     // G13   (bass G)
        { at: 2.5,  bass: 36, notes: [52, 55, 59, 62] },     // Cmaj9 (bass C)
        { at: 3.75, bass: 45, notes: [55, 60, 64, 67] }      // Am7   (bass A)
      ];
      chords.forEach(function (c) {
        var start = t0 + c.at, stop = Math.min(start + 1.45, END);
        c.notes.forEach(function (m, k) {
          var s = start + k * 0.018;                 // tiny strum
          var g = ctx.createGain();
          g.gain.setValueAtTime(0, s);
          g.gain.linearRampToValueAtTime(0.075, s + 0.02);
          g.gain.exponentialRampToValueAtTime(0.03, s + 0.7);
          g.gain.exponentialRampToValueAtTime(0.0008, stop);
          [["sine", 1, 1], ["triangle", 1, 0.35], ["sine", 2, 0.12]].forEach(function (p) {
            var o = ctx.createOscillator(), og = ctx.createGain();
            o.type = p[0]; o.frequency.value = hz(m) * p[1]; og.gain.value = p[2];
            wowDepth.connect(o.detune);
            o.connect(og); og.connect(g);
            o.start(s); o.stop(stop + 0.05);
          });
          g.connect(master);
        });
        // bass
        var b = ctx.createOscillator(), bg = ctx.createGain();
        b.type = "sine"; b.frequency.value = hz(c.bass);
        bg.gain.setValueAtTime(0, start);
        bg.gain.linearRampToValueAtTime(0.28, start + 0.03);
        bg.gain.exponentialRampToValueAtTime(0.08, start + 0.9);
        bg.gain.exponentialRampToValueAtTime(0.0008, stop);
        b.connect(bg); bg.connect(master); b.start(start); b.stop(stop + 0.05);
      });

      // ---- drums (positions in beats; off-beat eighths are swung) ----
      function kick(t) {
        var o = ctx.createOscillator(), g = ctx.createGain();
        o.frequency.setValueAtTime(115, t);
        o.frequency.exponentialRampToValueAtTime(42, t + 0.14);
        g.gain.setValueAtTime(0.75, t);
        g.gain.exponentialRampToValueAtTime(0.001, t + 0.38);
        o.connect(g); g.connect(master); o.start(t); o.stop(t + 0.4);
      }
      function snare(t) {
        var n = ctx.createBufferSource(); n.buffer = noise;
        var bp = ctx.createBiquadFilter(); bp.type = "bandpass"; bp.frequency.value = 1700; bp.Q.value = 0.8;
        var g = ctx.createGain();
        g.gain.setValueAtTime(0.32, t);
        g.gain.exponentialRampToValueAtTime(0.001, t + 0.2);
        n.connect(bp); bp.connect(g); g.connect(master);
        n.start(t, Math.random() * 2, 0.22);
        var body = ctx.createOscillator(), bg = ctx.createGain();
        body.frequency.value = 185;
        bg.gain.setValueAtTime(0.12, t);
        bg.gain.exponentialRampToValueAtTime(0.001, t + 0.09);
        body.connect(bg); bg.connect(master); body.start(t); body.stop(t + 0.1);
      }
      function hat(t, vol) {
        var n = ctx.createBufferSource(); n.buffer = noise;
        var hp = ctx.createBiquadFilter(); hp.type = "highpass"; hp.frequency.value = 7000;
        var g = ctx.createGain();
        g.gain.setValueAtTime(vol, t);
        g.gain.exponentialRampToValueAtTime(0.001, t + 0.05);
        n.connect(hp); hp.connect(g); g.connect(master);
        n.start(t, Math.random() * 2, 0.06);
      }
      [0, 1.75, 2.5, 4].forEach(function (b) { kick(t0 + b * BEAT); });
      [1, 3, 5].forEach(function (b) { snare(t0 + b * BEAT); });
      for (var e = 0; e < 12; e++) {
        var beat = Math.floor(e / 2) + (e % 2 ? 0.6 : 0);  // swing: off-beat lands late
        var t = t0 + beat * BEAT;
        if (t < END - 0.05) hat(t, e % 2 ? 0.035 : 0.06);
      }

      // ---- tape hiss (steady, quiet) ----
      var hiss = ctx.createBufferSource(); hiss.buffer = noise;
      var hf = ctx.createBiquadFilter(); hf.type = "bandpass"; hf.frequency.value = 5000; hf.Q.value = 0.3;
      var hg = ctx.createGain(); hg.gain.value = 0.02;
      hiss.connect(hf); hf.connect(hg); hg.connect(limit);  // skips the warm filter so it stays airy
      hiss.start(t0); hiss.stop(END);

      // ---- vinyl crackle (random soft pops) ----
      var crack = ctx.createBuffer(1, Math.floor(ctx.sampleRate * CLIP_S), ctx.sampleRate);
      var cd = crack.getChannelData(0);
      for (var j = 0; j < cd.length; j++) if (Math.random() < 0.00035) cd[j] = (Math.random() * 2 - 1) * 0.7;
      var cs = ctx.createBufferSource(); cs.buffer = crack;
      var cf = ctx.createBiquadFilter(); cf.type = "highpass"; cf.frequency.value = 1200;
      var cgn = ctx.createGain(); cgn.gain.setValueAtTime(0, t0);
      cgn.gain.linearRampToValueAtTime(0.35, t0 + 0.1);
      cgn.gain.setValueAtTime(0.35, END - 0.6);
      cgn.gain.linearRampToValueAtTime(0, END);
      cs.connect(cf); cf.connect(cgn); cgn.connect(limit);
      cs.start(t0);
    }
    window.lsBuildLofiClip = buildClip; // exposed so the clip can be tested offline

    document.querySelectorAll("[data-ls-skeleton]").forEach(function (root) {
      if (root.getAttribute("data-ls-ready")) return; // don't wire twice
      root.setAttribute("data-ls-ready", "1");

      var card = root.querySelector(".ls-skel-card");
      var loading = root.querySelector('[data-ls-state="loading"]');
      var loaded = root.querySelector('[data-ls-state="loaded"]');
      var art = root.querySelector(".ls-art");
      var playBtn = root.querySelector(".ls-play");
      var timer = null;
      var audio = null; // { ctx, end } while playing

      function stopClip() {
        if (!audio) return;
        var a = audio; audio = null;
        clearTimeout(a.end);
        try { a.ctx.close(); } catch (e) {}
        art.classList.remove("ls-playing");
        playBtn.setAttribute("aria-label", "Play 5-second lofistack radio clip");
      }

      function playClip() {
        var AC = window.AudioContext || window.webkitAudioContext;
        if (!AC) return; // very old browser: button just does nothing
        var ctx = new AC();
        buildClip(ctx, ctx.destination, ctx.currentTime + 0.05);
        audio = { ctx: ctx, end: setTimeout(stopClip, CLIP_S * 1000 + 150) };
        art.classList.remove("ls-playing"); void art.offsetWidth; // restart progress line
        art.classList.add("ls-playing");
        playBtn.setAttribute("aria-label", "Stop clip");
      }

      playBtn.addEventListener("click", function () {
        if (audio) stopClip(); else playClip();
      });

      function start() {
        stopClip();
        clearTimeout(timer);
        card.setAttribute("aria-busy", "true");
        loaded.hidden = true;
        loading.hidden = false;
        timer = setTimeout(function () {
          loading.hidden = true;
          // re-trigger the fade-in each time
          loaded.classList.remove("ls-loaded");
          void loaded.offsetWidth;
          loaded.classList.add("ls-loaded");
          loaded.hidden = false;
          card.setAttribute("aria-busy", "false");
        }, LOAD_MS);
      }

      root.querySelector(".ls-replay").addEventListener("click", start);
      start();
    });
  })();
</script>

# Final prompt — Skeleton Screen Loader

Build a skeleton screen loader as one self-contained block of plain HTML, CSS and JavaScript (no frameworks, no libraries, no audio files) that I can paste into a GoHighLevel custom code element.

Loading behavior:
- Show a placeholder card with the exact same shape as the real card: an image block (4:3), a title line, a shorter subtitle line, and a row with a small circle and a short line.
- Every placeholder block has a soft shimmer — a faint light band sweeping left to right on a ~1.4s loop, done in CSS only.
- Pretend to load for about 2.2 seconds, then swap to the real card with a short fade-and-rise (~400ms).
- Add a small "replay loading" button under the card that restarts the whole thing (and stops any music playing).
- Put both states in the HTML and toggle them with the `hidden` attribute (include a CSS rule so `hidden` always wins over the page builder's styles).

Play button and 5-second radio clip:
- On the real card's artwork, add a round amber play button (inline SVG play icon, switching to a pause icon while playing) next to the label "side b · 5 sec preview".
- Clicking it plays a 5-second original lofi clip generated live with the Web Audio API — nothing to download or host. The clip: jazzy chords Dm9 → G13 → Cmaj9 → Am7 (1.25s each) on a soft electric-piano tone (sine + a little triangle + a quiet octave), a tiny strum between notes, a slow "tape wobble" that bends the pitch a few cents, a round sine bass on each chord's root, a lazy swung drum beat at 72 BPM (soft kick, noise snare, quiet hi-hats with late off-beats), steady quiet tape hiss and random soft vinyl pops.
- Run everything through a warm low-pass filter and a limiter so it never clips; fade in quickly and fade out over the last second.
- A thin amber progress line fills along the bottom of the artwork over the 5 seconds. Clicking again stops it early. When it ends, the button resets.
- Only create the audio when the button is clicked (browsers block sound before a click). Give the button an `aria-label` that changes between "Play 5-second lofistack radio clip" and "Stop clip".

Accessibility: the card sets `aria-busy` while loading and `aria-live="polite"` so screen readers hear the change. Turn animations off if the visitor has "reduce motion" on.

Safety with the page builder: prefix every class with `ls-`, wrap the script in a function, and guard against wiring it twice.

Style: warm dark lofi palette — deep brown card (#282219), cream text (#ece4d6), muted tan secondary text (#a2977f), amber accent (#e0985f). The real card is a pretend "lofi radio" playlist: "rainy window study mix", "1 hr 12 min · tape hiss included", station "lofistack radio".
